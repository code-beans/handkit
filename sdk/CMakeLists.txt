cmake_minimum_required(VERSION 3.24)
project(
    handkit
    VERSION 0.4.0
    LANGUAGES C)
if(NOT TARGET SDL3::SDL3)
    find_package(SDL3 CONFIG REQUIRED)
endif()
set(HANDKIT_DEVICE "trimui" CACHE STRING "Handheld device integration")
set_property(CACHE HANDKIT_DEVICE PROPERTY STRINGS trimui arkos)
if(NOT HANDKIT_DEVICE MATCHES "^(trimui|arkos)$")
    message(FATAL_ERROR "Unknown HANDKIT_DEVICE: ${HANDKIT_DEVICE}")
endif()
if(HANDKIT_DEVICE STREQUAL "arkos")
    add_library(handkit STATIC src/runtime.c src/menu.c src/system_arkos.c)
    target_compile_definitions(handkit PUBLIC HANDKIT_DEVICE_ARKOS=1)
else()
    add_library(handkit STATIC src/runtime.c src/feedback.c src/menu.c src/system.c)
endif()
add_library(Handkit::Handkit ALIAS handkit)
target_link_libraries(handkit PUBLIC SDL3::SDL3 ${CMAKE_DL_LIBS})
target_include_directories(handkit PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
install(TARGETS handkit ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY cmake/ DESTINATION lib/cmake/Handkit)

option(HANDKIT_BUILD_TESTS "Build SDK checks" OFF)
if(HANDKIT_BUILD_TESTS AND HANDKIT_DEVICE STREQUAL "arkos")
    enable_testing()
    add_executable(handkit-arkos-check tests/arkos.c)
    target_link_libraries(handkit-arkos-check PRIVATE Handkit::Handkit)
    add_test(NAME handkit-arkos COMMAND handkit-arkos-check)
endif()
